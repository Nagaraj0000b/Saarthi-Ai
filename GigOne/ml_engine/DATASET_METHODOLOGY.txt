DATASET GENERATION METHODOLOGY
PSEUDOCODE FOR WORD DOCUMENT

--------------------------------------------------
ALGORITHM 1: DEFINE GLOBAL CONFIGURATION
--------------------------------------------------

SET TOTAL_ROWS = 50000
SET COLD_START_RATIO = 0.15
SET RANDOM_SEED = 42

DEFINE JOB_TYPES AS:
    "Ride hailing drivers (cab+ bike)" ->
        [Uber, Ola, BluSmart, Namma Yatri, InDriver, Rapido]
    "Ride hailing instant delivery" ->
        [Swiggy, Zomato, Blinkit, Zepto]
    "Delivery workers in General" ->
        [Amazon Flex, Delhivery, BlueDart, Dunzo, BigBasket, JioMart]
    "On Demand home based services" ->
        [Urban Company, Local Plumber, Local Electrician, Home Cleaning Co, Elderly Care, Pet Walker]
    "Remote service providers" ->
        [DataEntry Inc, SupportHero, Virtual Assistant Hub, Translation Pro]

DEFINE WEATHER_OPTIONS WITH PROBABILITIES AS:
    Clear -> 0.50
    Rain -> 0.12
    Drizzle -> 0.10
    Thunderstorm -> 0.05
    Clouds -> 0.15
    Haze -> 0.08

DEFINE TRAFFIC_OPTIONS WITH PROBABILITIES AS:
    clear -> 0.35
    moderate -> 0.40
    heavy -> 0.25

DEFINE SKILL_LIST AS:
    skill_bike, skill_car, skill_heavy,
    skill_clean, skill_trade, skill_care,
    skill_digital, skill_support

DEFINE BASE_RATES FOR EACH PLATFORM
DEFINE JOB_TO_TYPE MAPPING
DEFINE JOB_SKILL_COMPATIBILITY MAPPING


--------------------------------------------------
ALGORITHM 2: GENERATE ONE SYNTHETIC ROW
--------------------------------------------------

PROCEDURE GENERATE_ROW(job_name, is_cold_start)

    SET job_type = JOB_TO_TYPE[job_name]

    STEP 1: Generate time context
        hour = RANDOM_INTEGER between 0 and 23
        day = RANDOM_INTEGER between 0 and 6

    STEP 2: Generate weather context
        weather = RANDOM_CHOICE from WEATHER_OPTIONS using defined probabilities
        temp = ROUND(NORMAL_DISTRIBUTION(mean = 30, std = 5), 1)
        temp = CLAMP(temp, minimum = 15, maximum = 45)

    STEP 3: Generate traffic context
        traffic = RANDOM_CHOICE from TRAFFIC_OPTIONS using defined probabilities

        IF traffic = "clear" THEN
            congestion_percent = RANDOM_INTEGER between 0 and 14
        ELSE IF traffic = "moderate" THEN
            congestion_percent = RANDOM_INTEGER between 15 and 44
        ELSE
            congestion_percent = RANDOM_INTEGER between 45 and 99
        END IF

    STEP 4: Generate user skill profile
        FOR EACH skill IN SKILL_LIST DO
            ASSIGN skill value randomly as 0 or 1
        END FOR

        IF sum of all assigned skills = 0 THEN
            ASSIGN 1 to one randomly selected skill
        END IF

        required_skills = JOB_SKILL_COMPATIBILITY[job_name]
        is_compatible = TRUE if user has at least one required skill
        is_compatible = FALSE otherwise

    STEP 5: Generate worker history
        IF is_cold_start = TRUE THEN
            hours_today = 0.0
            hours_last_3_days = 0.0
            consecutive_days = 0
            mood_score = ROUND(UNIFORM_DISTRIBUTION(-0.2, 0.3), 2)
            burnout_risk = "low"
        ELSE
            hours_today = ROUND(EXPONENTIAL_DISTRIBUTION(scale = 3.0), 1)
            hours_today = MIN(hours_today, 14.0)

            hours_last_3_days = ROUND(EXPONENTIAL_DISTRIBUTION(scale = 10.0), 1)
            hours_last_3_days = MIN(hours_last_3_days, 42.0)

            consecutive_days = INTEGER(EXPONENTIAL_DISTRIBUTION(scale = 3.0))
            consecutive_days = MIN(consecutive_days, 21)

            mood_score = ROUND(NORMAL_DISTRIBUTION(mean = 0.0, std = 0.4), 2)
            mood_score = CLAMP(mood_score, minimum = -1.0, maximum = 1.0)

            risk_value =
                (hours_today * 5) +
                (hours_last_3_days * 2) +
                (consecutive_days * 3) -
                (mood_score * 20)

            IF risk_value > 70 THEN
                burnout_risk = "high"
            ELSE IF risk_value > 40 THEN
                burnout_risk = "moderate"
            ELSE
                burnout_risk = "low"
            END IF
        END IF

    STEP 6: Initialize earning rate
        rate = BASE_RATES[job_name]

    STEP 7: Apply weather impact
        IF job_type != "Remote service providers" THEN
            is_rain = TRUE if weather is Rain or Thunderstorm
            is_drizzle = TRUE if weather is Drizzle
            is_heat = TRUE if temp > 36

            IF is_rain = TRUE THEN
                IF job_type = "Ride hailing drivers (cab+ bike)" THEN
                    rate = rate * NORMAL_DISTRIBUTION(mean = 1.35, std = 0.08)
                ELSE IF job_type = "Ride hailing instant delivery" THEN
                    rate = rate * NORMAL_DISTRIBUTION(mean = 1.15, std = 0.05)
                ELSE IF job_type = "Delivery workers in General" THEN
                    rate = rate * NORMAL_DISTRIBUTION(mean = 0.85, std = 0.05)
                ELSE IF job_type = "On Demand home based services" THEN
                    rate = rate * NORMAL_DISTRIBUTION(mean = 0.90, std = 0.10)
                END IF
            ELSE IF is_drizzle = TRUE THEN
                IF job_type = "Ride hailing drivers (cab+ bike)" THEN
                    rate = rate * NORMAL_DISTRIBUTION(mean = 1.15, std = 0.05)
                END IF
            END IF

            IF is_heat = TRUE THEN
                IF job_type is in ["Food Delivery", "Quick Commerce"] THEN
                    rate = rate * NORMAL_DISTRIBUTION(mean = 1.18, std = 0.05)
                END IF
            END IF
        END IF

    STEP 8: Apply traffic impact
        IF job_type != "Remote service providers" THEN
            IF traffic = "heavy" THEN
                IF job_type = "Ride hailing drivers (cab+ bike)" THEN
                    rate = rate * NORMAL_DISTRIBUTION(mean = 0.72, std = 0.06)
                ELSE IF job_type = "Delivery workers in General" THEN
                    rate = rate * NORMAL_DISTRIBUTION(mean = 0.78, std = 0.05)
                ELSE IF job_type = "Ride hailing instant delivery" THEN
                    rate = rate * NORMAL_DISTRIBUTION(mean = 1.12, std = 0.05)
                END IF
            ELSE IF traffic = "moderate" THEN
                IF job_type = "Ride hailing drivers (cab+ bike)" THEN
                    rate = rate * NORMAL_DISTRIBUTION(mean = 0.92, std = 0.04)
                END IF
            END IF
        END IF

    STEP 9: Apply time-based demand multipliers
        IF job_type = "Ride hailing instant delivery" THEN
            IF hour is between 12 and 14 OR hour is between 19 and 22 THEN
                rate = rate * NORMAL_DISTRIBUTION(mean = 1.35, std = 0.07)
            END IF
        ELSE IF job_type = "Ride hailing drivers (cab+ bike)" THEN
            IF hour is between 8 and 10 OR hour is between 17 and 19 THEN
                rate = rate * NORMAL_DISTRIBUTION(mean = 1.25, std = 0.06)
            END IF
        END IF

        IF day >= 5 THEN
            IF job_type = "Ride hailing instant delivery" THEN
                rate = rate * NORMAL_DISTRIBUTION(mean = 1.15, std = 0.04)
            END IF
        END IF

    STEP 10: Apply worker fatigue penalty
        IF burnout_risk = "high" THEN
            rate = rate * NORMAL_DISTRIBUTION(mean = 0.68, std = 0.06)
        ELSE IF burnout_risk = "moderate" THEN
            rate = rate * NORMAL_DISTRIBUTION(mean = 0.88, std = 0.04)
        END IF

    STEP 11: Apply compatibility penalty
        IF is_compatible = FALSE THEN
            rate = MAX(0, NORMAL_DISTRIBUTION(mean = 10, std = 5))
        END IF

    STEP 12: Add final noise
        noise_factor = NORMAL_DISTRIBUTION(mean = 1.0, std = 0.15)
        target_earning = ROUND(MAX(0, rate * noise_factor), 2)

    STEP 13: Return completed row
        RETURN:
            job
            job_type
            hour
            day
            weather
            temp
            traffic
            congestion_percent
            is_compatible
            hours_today
            hours_last_3_days
            consecutive_days
            mood_score
            burnout_risk
            target_earning
            all skill columns

END PROCEDURE


--------------------------------------------------
ALGORITHM 3: GENERATE FULL DATASET
--------------------------------------------------

PROCEDURE MAIN

    CREATE output directory if it does not exist

    SET ROWS_PER_CATEGORY = 10000
    INITIALIZE data_rows as empty list

    FOR EACH category_name, jobs_list IN JOB_TYPES DO

        rows_per_job = INTEGER_DIVISION(ROWS_PER_CATEGORY, number of jobs in jobs_list)
        remainder = MODULO(ROWS_PER_CATEGORY, number of jobs in jobs_list)

        FOR EACH job_name WITH index IN jobs_list DO

            current_job_rows = rows_per_job
            IF index < remainder THEN
                current_job_rows = current_job_rows + 1
            END IF

            cold_rows_count = INTEGER(current_job_rows * COLD_START_RATIO)
            normal_rows_count = current_job_rows - cold_rows_count

            REPEAT cold_rows_count times
                APPEND GENERATE_ROW(job_name, TRUE) to data_rows
            END REPEAT

            REPEAT normal_rows_count times
                APPEND GENERATE_ROW(job_name, FALSE) to data_rows
            END REPEAT

        END FOR

    END FOR

    CONVERT data_rows into dataframe
    SHUFFLE all rows randomly
    SAVE dataframe as synthetic_earnings.csv

END PROCEDURE


--------------------------------------------------
KEY DESIGN RULES
--------------------------------------------------

1. Total dataset size is 50,000 rows.
2. Exactly 15% of rows per platform represent cold-start users.
3. Each job category contributes 10,000 rows.
4. Weather, traffic, workload, and mood are generated probabilistically.
5. At least one skill is always assigned to every synthetic user.
6. Incompatible users are forced toward near-zero earnings.
7. Final earnings always include random noise for realism.
